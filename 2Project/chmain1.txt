    <script>
      function validateForm() {
        const form = document.forms["contactForm"];
        const name = form["Your Name"].value;
        const email = form["Email"].value;
        const phone = form["Phone"].value;

        if (!name) {
          alert("名前が未入力です");
          form["Your Name"].focus();
          return false;
        }
        if (!email) {
          alert("メールが未入力です");
          form["Email"].focus();
          return false;
        }
        if (!phone) {
          alert("電話番号が未入力です");
          form["Phone"].focus();
          return false;
        }
        return true;
      }

      // ユーザーが自分の安否情報を見ている場合だけ編集ボタン表示（仮）
      const isOwnData = true; // 本来はサーバーから取得
      if (isOwnData) {
        document.getElementById("edit-button").style.display = "inline-block";
      }
    </script>
